def _budget(deadline: float) -> float:
    return max(0.0, deadline - time.time())

async def _wait_for_floor(coro, seconds_cap: float, floor_seconds: float, deadline: float):
    """
    남은 예산이 floor보다 작아도 floor 만큼은 대기한다.
    단, 외부에서 전체 타임아웃으로 끊길 수 있으므로 floor는 과도하게 크게 잡지 말 것.
    """
    t_cap = min(seconds_cap, seconds_cap)  # 의미상 cap 유지
    t = max(floor_seconds, min(t_cap, _budget(deadline)))
    return await asyncio.wait_for(coro, timeout=t)

async def _wait_for(coro, seconds: float, deadline: float):
    t = min(seconds, _budget(deadline))
    if t <= 0.05:
        raise asyncio.TimeoutError("budget_exhausted")
    return await asyncio.wait_for(coro, timeout=t)

def compute_date_finish_info(date_finish_str: str) -> str:
    if not date_finish_str: return ""
    try:
        seoul = pytz.timezone("Asia/Seoul")
        naive = datetime.strptime(date_finish_str.strip(), "%Y.%m.%d %H:%M")
        finish_dt = seoul.localize(naive)
        now_k = datetime.now(seoul)
        diff = now_k - finish_dt
        if diff.total_seconds() < 0: return ""
        if diff.total_seconds() < 60: return "방금 전 종료"
        total_min = int(diff.total_seconds() // 60)
        h, m = divmod(total_min, 60)
        if h >= 24:
            d = h // 24; rh = h % 24
            return f"{d}일 {rh}시간 전 종료" if rh else f"{d}일 전 종료"
        return f"{h}시간 {m}분 전 종료" if h else f"{m}분 전 종료"
    except Exception:
        return ""

def parse_status(status_text: str) -> str:
    text = (status_text or "").strip()
    m = re.search(r"(?:(\d+)\s*시간)?\s*(?:(\d+)\s*분)?\s*(?:충전중|사용중)", text) or \
        re.search(r"(?:충전중|사용중)\s*(?:(\d+)\s*시간)?\s*(?:(\d+)\s*분)?", text)
    if m:
        hh, mm = m.group(1), m.group(2)
        base = "충전중" if "충전중" in text else ("사용중" if "사용중" in text else text)
        h = int(hh) if hh else 0
        mv = int(mm) if mm else 0
        if not hh and mv >= 60:
            h2, r2 = divmod(mv, 60)
            h += h2; mv = r2
        if h and mv: return f"{base}\n({h}시간 {mv}분)"
        if h:        return f"{base}\n({h}시간)"
        if mv:       return f"{base}\n({mv}분)"
        return base + "\n"
    return text + "\n"

def _json_loose_parse(txt: str) -> Optional[Any]:
    txt = (txt or "").strip()
    if not txt: return None
    # JSONP
    if "(" in txt and txt.endswith(")"):
        m = re.search(r"\(\s*({[\s\S]+})\s*\)\s*;?\s*$", txt)
        if m: txt = m.group(1)
    # var a = {...}
    m = re.search(r"=\s*({[\s\S]+})\s*;?$", txt)
    if m: txt = m.group(1)
    try:
        return json.loads(txt)
    except Exception:
        m = re.search(r"({[\s\S]+})", txt)
        if m:
            try: return json.loads(m.group(1))
            except Exception: return None
        return None

def _extract_items_from_any(data: Any) -> Optional[List[dict]]:
    paths=[]
    def walk(node, path=""):
        if isinstance(node, list) and node and isinstance(node[0], dict):
            paths.append((path, node))
        elif isinstance(node, dict):
            for k,v in node.items():
                walk(v, f"{path}.{k}" if path else k)
    walk(data)
    for key in ("chargers","items","rows","list","data","result"):
        for p, arr in paths:
            if p.endswith(key): return arr
    return paths[0][1] if paths else None

def _direct_http(url: str, method: str="GET", headers: Optional[Dict[str,str]]=None, body: Optional[str]=None, timeout: float=8.0) -> Tuple[int, str, Dict[str,str]]:
    req = urllib.request.Request(url=url, method=method)
    if headers:
        for k,v in headers.items():
            if k.lower() in ("host","content-length"): continue
            req.add_header(k, v)
    data = body.encode("utf-8") if (body is not None) else None
    try:
        with urllib.request.urlopen(req, data=data, timeout=timeout) as resp:
            code = resp.getcode()
            ctype = resp.headers.get("Content-Type","")
            txt = resp.read().decode("utf-8","ignore")
            return code, txt, {"content-type":ctype}
    except Exception as e:
        return 0, str(e), {}

    if not url_tmpl: return None
    url  = url_tmpl.replace("{sid}", urllib.parse.quote_plus(sid))
    body = body_tmpl.replace("{sid}", sid) if body_tmpl else None
    code, text, resp_headers = _direct_http(url, method, headers, body, DIRECT_TIMEOUT_S)
    if code and 200 <= code < 300 and text:
        data = _json_loose_parse(text) or {}
        items = _extract_items_from_any(data) or []
        mapped = map_to_output(items)
        if mapped:
            return mapped
    return None

def _maybe_learn_hint_from_logs(sid: str, xhr_logs: List[dict], host: str):
    if not AUTO_DIRECT or not xhr_logs: return
    cand = None
    for l in reversed(xhr_logs):
        u = l.get("url","") or ""
        b = l.get("body","") or ""
        if sid and (sid in u or sid in b):
            cand = l; break
    if not cand: return
    url_tmpl = _sidize_template(cand.get("url",""), sid)
    body_tmpl= _sidize_template((cand.get("body","") or ""), sid)
    if not url_tmpl and not body_tmpl: return
    headers = {}
    ct = cand.get("ct","")
    if "x-www-form-urlencoded" in (ct or ""):
        headers["Content-Type"] = "application/x-www-form-urlencoded; charset=UTF-8"
    hints = _load_hints()
    hints[host] = {
        "url_tmpl": url_tmpl or "",
        "method":   (cand.get("method","GET") or "GET").upper(),
        "headers":  headers,
        "body_tmpl":body_tmpl or ""
    }
    _save_hints(hints)
    log_info("learned auto-direct template for %s", host)

class ChromeDaemon:
    def __init__(self):
        self.proc = None
        self.profile_dir = None
        self.ws_endpoint = None

    def _cached_ws(self) -> Optional[str]:
        try:
            if os.path.exists(ACTIVE_EP):
                ep = open(ACTIVE_EP,"r").read().strip()
                if ep.startswith("ws://"): return ep
        except Exception: pass
        ap = os.path.join(ACTIVE_DIR,"DevToolsActivePort")
        try:
            if os.path.exists(ap):
                with open(ap,"r") as f:
                    lines=[x.strip() for x in f if x.strip()]
                if lines:
                    port = lines[0]; suf = lines[1] if len(lines)>1 else "/devtools/browser"
                    return f"ws://127.0.0.1:{port}{suf if suf.startswith('/devtools') else '/devtools/browser'}"
        except Exception: pass
        return None

    async def connect_existing(self, deadline: float):
        ws = self._cached_ws()
        if not ws: return None
        log_info("try attach existing daemon %s", ws)
        try:
            br = await _wait_for_floor(connect(browserWSEndpoint=ws), 6.0, 4.0, deadline)
            self.ws_endpoint = ws; self.profile_dir = ACTIVE_DIR
            log_info("attached existing daemon")
            return br
        except Exception as e:
            log_warn("attach failed %s", repr(e))
            return None

    async def _hard_kill_all(self):
        try:
            if shutil.which("pkill"):
                p=await asyncio.create_subprocess_exec("pkill","-9","-f","chromium", stdout=asyncio.subprocess.DEVNULL, stderr=asyncio.subprocess.DEVNULL)
                await p.wait()
            elif shutil.which("killall"):
                p=await asyncio.create_subprocess_exec("killall","-9","chromium", stdout=asyncio.subprocess.DEVNULL, stderr=asyncio.subprocess.DEVNULL)
                await p.wait()
        except Exception:
            pass
        await asyncio.sleep(0.3)

    async def start(self, deadline: float) -> bool:
        os.makedirs(ACTIVE_DIR, exist_ok=True)
        cache_dir="/tmp/chrome-cache"; os.makedirs(cache_dir, exist_ok=True)
        self.profile_dir = ACTIVE_DIR

        args = [
            CHROME_BIN,
            "--headless=new",
            "--no-sandbox","--disable-setuid-sandbox",
            "--remote-debugging-port=0",
            f"--user-data-dir={self.profile_dir}",
            "--enable-logging=stderr","--v=1",
            "--disable-gpu","--disable-extensions","--no-zygote","--no-first-run",
            "--disable-background-networking","--disable-default-apps","--disable-sync",
            "--metrics-recording-only","--mute-audio",
            "--blink-settings=imagesEnabled=false",
            "--proxy-server=direct://","--proxy-bypass-list=*",
            "--disable-features=IsolateOrigins,site-per-process",
            f"--disk-cache-dir={cache_dir}",
            "--disable-software-rasterizer",
            "--disable-dev-shm-usage",
        ]

        async def _spawn():
            self.proc = await asyncio.create_subprocess_exec(
                *args,
                stdout=asyncio.subprocess.DEVNULL,
                stderr=asyncio.subprocess.PIPE
            )
            ws_re = re.compile(r"DevTools listening on (ws://[^\s]+)")
            active_port = os.path.join(self.profile_dir,"DevToolsActivePort")
            # ★ 런치 대기는 '최소 보장 시간' 사용
            end = time.time() + max(MIN_LAUNCH_FLOOR_S, min(LAUNCH_TIMEOUT_S, _budget(deadline)))
            while time.time() < end:
                try:
                    chunk = await asyncio.wait_for(self.proc.stderr.readline(), timeout=0.3)
                    if chunk:
                        line = chunk.decode("utf-8","ignore").rstrip("\n")
                        _w(line)
                        m = ws_re.search(line)
                        if m:
                            self.ws_endpoint = m.group(1)
                            open(ACTIVE_EP,"w").write(self.ws_endpoint)
                            log_info("daemon ws endpoint %s", self.ws_endpoint)
                            return True
                except asyncio.TimeoutError:
                    pass
                if os.path.exists(active_port):
                    try:
                        with open(active_port,"r") as f:
                            lines=[x.strip() for x in f if x.strip()]
                        if lines:
                            port=lines[0]; suf=lines[1] if len(lines)>1 else "/devtools/browser"
                            self.ws_endpoint = f"ws://127.0.0.1:{port}{suf if suf.startswith('/devtools') else '/devtools/browser'}"
                            open(ACTIVE_EP,"w").write(self.ws_endpoint)
                            log_info("daemon ws endpoint %s", self.ws_endpoint)
                            return True
                    except Exception as e:
                        log_warn("ActivePort parse fail %s", repr(e))
            return False

        log_info("starting daemon chromium (persistent)")
        ok = await _spawn()
        if ok: return True

        # 1차 실패 → 하드킬 후 재시도
        log_warn("daemon launch timeout -> hard kill & retry")
        await self._hard_kill_all()
        ok = await _spawn()
        if ok: return True

        # 2차 실패 → 강제 종료(KEEP_DAEMON 무시)
        log_warn("daemon launch failed twice -> force stop")
        await self.stop(force=True)
        return False

    async def stop(self, force: bool=False):
        if KEEP_DAEMON and not force:
            log_info("KEEP_DAEMON=1 -> keep chromium running")
            return
        try:
            if self.proc and self.proc.returncode is None:
                self.proc.send_signal(signal.SIGKILL)
                try:
                    await asyncio.wait_for(self.proc.wait(), timeout=2.0)
                except asyncio.TimeoutError:
                    pass
        except Exception:
            pass
        # ActivePort/endpoint 파일 정리(낡은 포트로 연결 방지)
        try:
            for p in (ACTIVE_EP, os.path.join(ACTIVE_DIR,"DevToolsActivePort")):
                if os.path.exists(p): os.remove(p)
        except Exception:
            pass

async def _get_page_with_restart(daemon: ChromeDaemon, browser, deadline: float):
    """
    1) 기존 탭 재사용 → 2) 기본/시크릿 동시 시도 → 3) 브라우저 재연결 →
    4) 데몬 강제 재기동 후 최종 시도. 모두 실패하면 TimeoutError.
    모든 newPage/연결에는 '최소 보장 시간' 적용.
    """
    async def _ping(page):
        try:
            await _wait_for_floor(page.evaluate("()=>'ok'"), 2.0, 1.0, deadline)
            return True
        except Exception:
            return False

    # 1) 기존 탭 재사용
    try:
        pages = await _wait_for_floor(browser.pages(), 5.0, 2.0, deadline)
        for p in pages:
            if await _ping(p):
                log_info("reuse existing page")
                return (None, p)
    except Exception:
        pass

    # 2) 기본/시크릿 동시 시도
    async def _default():
        p = await _wait_for_floor(browser.newPage(), NEWPAGE_TIMEOUT_S, MIN_NEWPAGE_FLOOR_S, deadline)
        if not await _ping(p): raise asyncio.TimeoutError("page_ping_fail")
        return (None, p)
    async def _incog():
        ctx = await _wait_for_floor(browser.createIncognitoBrowserContext(), NEWPAGE_TIMEOUT_S, MIN_NEWPAGE_FLOOR_S, deadline)
        p   = await _wait_for_floor(ctx.newPage(), NEWPAGE_TIMEOUT_S, MIN_NEWPAGE_FLOOR_S, deadline)
        if not await _ping(p): raise asyncio.TimeoutError("page_ping_fail")
        return (ctx, p)

    try:
        done, pending = await asyncio.wait(
            {asyncio.create_task(_default()), asyncio.create_task(_incog())},
            timeout=max(MIN_NEWPAGE_FLOOR_S, min(NEWPAGE_TIMEOUT_S, _budget(deadline))),
            return_when=asyncio.FIRST_COMPLETED
        )
        for t in pending: t.cancel()
        if done:
            return list(done)[0].result()
    except Exception as e:
        log_warn("newPage race failed (%s), will reconnect", repr(e))

    # 3) 브라우저 재연결
    try:
        ws = daemon._cached_ws()
        if ws:
            log_info("reconnect to daemon %s", ws)
            browser = await _wait_for_floor(connect(browserWSEndpoint=ws), 8.0, 4.0, deadline)
            p = await _wait_for_floor(browser.newPage(), NEWPAGE_TIMEOUT_S, MIN_NEWPAGE_FLOOR_S, deadline)
            if await _ping(p):
                return (None, p)
    except Exception as e:
        log_warn("reconnect/newPage failed %s", repr(e))

    # 4) 데몬 강제 재기동 (KEEP_DAEMON와 무관)
    log_warn("restart chromium daemon to recover")
    await daemon.stop(force=True)
    if not await daemon.start(deadline):
        raise asyncio.TimeoutError("daemon_restart_fail")
    try:
        browser = await _wait_for_floor(connect(browserWSEndpoint=daemon.ws_endpoint), 10.0, 6.0, deadline)
        p = await _wait_for_floor(browser.newPage(), NEWPAGE_TIMEOUT_S, MIN_NEWPAGE_FLOOR_S, deadline)
        if await _ping(p):
            return (None, p)
    except Exception as e:
        log_warn("post-restart newPage failed %s", repr(e))
    raise asyncio.TimeoutError("newPage_unavailable")

async def _browser_capture_and_parse(sid: str, deadline: float) -> dict:
    host = "www.ev.or.kr"

    daemon = ChromeDaemon()
    browser = await daemon.connect_existing(deadline)
    if not browser:
        ok = await daemon.start(deadline)
        if not ok:
            await daemon.stop(force=True)
            return {"msg":"RETRY","reason":"chrome_warming_up"}
        try:
            browser = await _wait_for_floor(connect(browserWSEndpoint=daemon.ws_endpoint), 8.0, 5.0, deadline)
        except Exception:
            await daemon.stop(force=True)
            return {"msg":"RETRY","reason":"chrome_connect_fail"}

    page=None; ctx=None
    xhr_json_blobs: List[Tuple[str, Any]] = []
    xhr_logs: List[Dict[str, Any]] = []

    try:
        log_info("acquire page")
        ctx, page = await _get_page_with_restart(daemon, browser, deadline)

        # inject & 환경 세팅
        await page.evaluateOnNewDocument(INJECT_JS)
        await page.setUserAgent(MOBILE_UA)
        await page.setViewport({"width": 390, "height": 844, "isMobile": True})
        await page.setExtraHTTPHeaders({
            "Accept-Language":"ko-KR,ko;q=0.9,en-US;q=0.8,en;q=0.7",
            "Cache-Control":"no-cache",
            "Referer":"https://www.ev.or.kr/"
        })
        await page.setRequestInterception(True)
        # 리소스 절감
        async def on_req(req):
            try:
                rt = getattr(req, "resourceType", None)
                if rt in ("image","font","media"):
                    await req.abort()
                else:
                    await req.continue_()
            except Exception:
                try: await req.continue_()
                except Exception: pass
        page.on("request", lambda r: asyncio.create_task(on_req(r)))

        async def on_resp(resp):
            try:
                ct=(resp.headers.get("content-type","") or "").lower()
                rt=getattr(getattr(resp,"request",None),"resourceType",None)
                if rt in ("xhr","fetch"):
                    txt=""
                    try: txt=await resp.text()
                    except: pass
                    if "application/json" in ct:
                        try:
                            xhr_json_blobs.append((resp.url, json.loads(txt)))
                        except: pass
                    else:
                        data=_json_loose_parse(txt)
                        if isinstance(data,(dict,list)):
                            xhr_json_blobs.append((resp.url, data))
                    xhr_logs.append({"url": resp.url, "status": resp.status, "ct": ct, "snippet": (txt or "")[:600]})
            except Exception:
                pass
        page.on("response", lambda r: asyncio.create_task(on_resp(r)))

        base_url = "https://www.ev.or.kr/nportal/monitor/evMapInfo.do?sid=%s&pFlag=Y" % urllib.parse.quote_plus(sid)

        # 1차 네비게이션
        nav1 = int(min(NAV_TIMEOUT1_MS, max(1500, _budget(deadline)*1000*0.6)))
        log_info("goto1 %s (timeout=%dms)", base_url, nav1)
        try:
            await _wait_for(page.goto(base_url, {"waitUntil":"domcontentloaded","timeout":nav1}), nav1/1000.0, deadline)
        except Exception as e:
            log_warn("goto1 failed %s", type(e).__name__)

        try:
            await _wait_for(page.waitForFunction("() => (window.__cap && window.__cap.count>=1)", {"polling":250}), 5.0, deadline)
        except Exception:
            pass

        # poll1
        end1 = time.time() + min(POLL_WINDOW_SEC, max(0.8, _budget(deadline)-2.5))
        while time.time() < end1 and _budget(deadline) > 0.5:
            try:
                cap = await _wait_for(page.evaluate(
                    "() => ({cnt:(window.__cap&&window.__cap.count)||0, logs:(window.__cap&&window.__cap.logs)||[], jsons:(window.__cap&&window.__cap.jsons)||[]})"
                ), 1.0, deadline)
                for j in cap.get("jsons", [])[:10]:
                    xhr_json_blobs.append((j.get("url",""), j.get("data")))
                for l in cap.get("logs", [])[:20]:
                    if isinstance(l, dict):
                        xhr_logs.append({k:l.get(k) for k in ("type","url","status","ct","snippet","method","body")})
            except Exception:
                pass

            for _url, data in list(xhr_json_blobs):
                items = _extract_items_from_any(data)
                if items:
                    mapped = map_to_output(items)
                    if mapped:
                        _maybe_learn_hint_from_logs(sid, xhr_logs, host)
                        mapped.setdefault("debug", {})["xhr_samples"] = xhr_logs[:12]
                        return mapped
            await asyncio.sleep(POLL_INTERVAL_SEC)

        # 2차 네비게이션
        nav2 = int(min(NAV_TIMEOUT2_MS, max(1500, _budget(deadline)*1000*0.6)))
        log_info("goto2 %s (timeout=%dms)", base_url, nav2)
        try:
            await _wait_for(page.goto(base_url, {"waitUntil":"networkidle2","timeout":nav2}), nav2/1000.0, deadline)
        except Exception as e:
            log_warn("goto2 failed %s", type(e).__name__)

        # poll2
        end2 = time.time() + min(POLL_WINDOW_SEC, max(0.8, _budget(deadline)-1.0))
        while time.time() < end2 and _budget(deadline) > 0.4:
            try:
                cap = await _wait_for(page.evaluate(
                    "() => ({cnt:(window.__cap&&window.__cap.count)||0, logs:(window.__cap&&window.__cap.logs)||[], jsons:(window.__cap&&window.__cap.jsons)||[]})"
                ), 1.0, deadline)
                for j in cap.get("jsons", [])[:10]:
                    xhr_json_blobs.append((j.get("url",""), j.get("data")))
                for l in cap.get("logs", [])[:20]:
                    if isinstance(l, dict):
                        xhr_logs.append({k:l.get(k) for k in ("type","url","status","ct","snippet","method","body")})
            except Exception:
                pass

            for _url, data in list(xhr_json_blobs):
                items = _extract_items_from_any(data)
                if items:
                    mapped = map_to_output(items)
                    if mapped:
                        _maybe_learn_hint_from_logs(sid, xhr_logs, host)
                        mapped.setdefault("debug", {})["xhr_samples"] = xhr_logs[:12]
                        return mapped
            await asyncio.sleep(POLL_INTERVAL_SEC)

        return {
            "msg":"NO_DATA","reason":"xhr_not_found",
            "title":"", "company_name":"",
            "total_chargers":0, "used_chargers":0, "remaining_chargers":0,
            "address":"", "chargers_info":[], "printString":"",
            "debug":{"xhr_samples": xhr_logs[:20]}
        }

    finally:
        try:
            if page: await page.close()
        except Exception: pass
        try:
            if ctx: await ctx.close()
        except Exception: pass
        # 데몬은 KEEP_DAEMON=1이면 유지

async def fetch_once(sid: str, overall_timeout: int = OVERALL_TIMEOUT_S) -> dict:
    started = time.time()
    deadline = started + overall_timeout - 1.0
    host = "www.ev.or.kr"

    # 1) 수동 다이렉트
    direct = _try_direct_manual(sid)
    if direct:
        direct["total_time"] = f"{time.time()-started:.2f} seconds"
        return direct

    # 2) 자동 다이렉트
    auto = _try_direct_auto(sid, host)
    if auto:
        auto["total_time"] = f"{time.time()-started:.2f} seconds"
        return auto

    # 3) 브라우저 → 캡처/파싱
    out = await _browser_capture_and_parse(sid, deadline)
    out["total_time"] = f"{time.time()-started:.2f} seconds"
    return out

def parse_args(argv: List[str]):
    sid=""; timeout=OVERALL_TIMEOUT_S; logflag=False
    for a in argv:
        if a.startswith("--sid="): sid=a.split("=",1)[1].strip()
        elif a.startswith("--timeout="):
            try: timeout=int(a.split("=",1)[1])
            except: pass
        elif a.startswith("--log="):
            v=a.split("=",1)[1].strip().lower(); logflag = v in ("1","true","yes","y")
    return sid, timeout, logflag

if __name__ == "__main__":
    sid, timeout, logflag = parse_args(sys.argv[1:])
    if logflag: LOG_LEVEL="DEBUG"
    if not sid:
        print(json.dumps({"msg":"NO_SID","error":"sid required"}, ensure_ascii=False), end=""); sys.exit(2)

    try:
        loop = asyncio.get_event_loop()
    except Exception:
        loop = asyncio.new_event_loop(); asyncio.set_event_loop(loop)

    try:
        payload = loop.run_until_complete(fetch_once(sid, overall_timeout=timeout))

        print(json.dumps(payload, ensure_ascii=False), end="")
        # 종료 코드
        code = 0 if payload.get("msg")=="SUCCESS" else (4 if payload.get("msg")=="RETRY" else (3 if payload.get("msg")=="NO_DATA" else 1))
        sys.exit(code)
    except Exception as e:
        log_err("fatal %s", repr(e))
        print(json.dumps({"msg":"ERROR","error":str(e)}, ensure_ascii=False), end="")

        $json = json_encode($data);

            $code = hexdec($m[1]);

            if ($code < 0x800) return chr(0xC0|($code>>6)).chr(0x80|($code&0x3F));

function cache_read($file, $ttl) {
    if (is_readable($file)) {
        $age = time() - filemtime($file);
        if ($age >= 0 && $age <= $ttl) {
            $txt = @file_get_contents($file);
            if ($txt !== false && $txt !== '') {
                $j = json_decode($txt, true);
                if (is_array($j) && isset($j['msg'])) return $j;
            }
        }
    }
    return null;
}
function cache_write($file, $arr) {
    @file_put_contents($file, json_encode_utf8($arr));
}

/* ---------- 캐시 히트 시 즉시 반환 ---------- */
$cached = cache_read($cache_file, $CACHE_TTL);
if ($cached && isset($cached['msg']) && $cached['msg']==='SUCCESS') {
    $cached['_cache'] = 'HIT';
    json_out($cached, 200);
}

/* ---------- docker exec 명령 구성 ---------- */
$container = 'pyppeteer-service';
$timeout_s = 55; // 바깥 타임아웃( /usr/bin/timeout ) — script.py는 45 내외로 완료되도록 구성
$cmd_base = 'docker exec ' . escapeshellarg($container)
         . ' env ' . ($debug ? 'LOG_LEVEL=DEBUG ' : '')
         . ' python3 /app/script.py'
         . ' --sid=' . escapeshellarg($sid)
         . ' --timeout=' . (int)$timeout_s;
if ($debug) $cmd_base .= ' --log=1';

if (is_executable('/usr/bin/timeout')) {
    $cmd_base = '/usr/bin/timeout ' . (int)($timeout_s + 5) . 's ' . $cmd_base;
}

/* ---------- 리트라이 정책 ---------- */
/*
  - 1차 호출: 성공 시 즉시 캐시/반환
  - 실패(RETQ/NO_DATA/ERROR/바깥 타임아웃 124) 시 200ms 후 1회 재호출
  - 둘 다 실패면 마지막 응답 반환 (HTTP 코드 매핑)
*/
$attempts = 2;
$last = null;

for ($i=1; $i<=$attempts; $i++) {
    $cmd = $cmd_base;

    $descriptors = array(
        0 => array('pipe','r'),
        1 => array('pipe','w'),
        2 => array('pipe','w'),
    );
    $proc = proc_open($cmd, $descriptors, $pipes);
    if (!is_resource($proc)) {
        $last = array('msg'=>'ERROR','error'=>'실행 시작 실패','cmd'=>$cmd);
        break;
    }
    fclose($pipes[0]);

    // stdout/stderr 수집
    $stdout = stream_get_contents($pipes[1]); fclose($pipes[1]);
    $stderr = stream_get_contents($pipes[2]); fclose($pipes[2]);
    $status = proc_close($proc);
    $body   = trim($stdout !== '' ? $stdout : $stderr);

    // JSON 파싱 시도
    $json = null;
    if ($body !== '' && ($body[0] === '{' || $body[0] === '[')) {
        $json = json_decode($body, true);
    }

    if (is_array($json) && isset($json['msg'])) {
        if ($json['msg'] === 'SUCCESS') {
            cache_write($cache_file, $json);
            send_status(200);
            header('Content-Type: application/json; charset=utf-8');
            echo json_encode_utf8($json);
            exit;
        }
        // 실패면 다음 시도 (마지막이면 탈출)
        $last = $json;
    } else {
        // 비정상 본문 처리 강화(바깥 타임아웃 포함)
        if ($status === 124) {
            // /usr/bin/timeout 에 의해 docker exec 전체가 타임아웃된 케이스
            $last = array(
                'msg'    => 'RETRY',
                'reason' => 'outer_timeout',
                'error'  => 'command timed out',
                'exit'   => 124
            );
        } else {
            $last = array(
                'msg'     => 'ERROR',
                'error'   => '비정상 응답 본문',
                'exit'    => $status,
                'snippet' => mb_substr($body, 0, 500, 'UTF-8')
            );
        }
    }

    // 재시도 간 짧은 휴지
    if ($i < $attempts) { usleep(200 * 1000); }
}

/* ---------- 최종 실패 응답 ---------- */
$http = 500;
if (is_array($last)) {
    if (isset($last['msg']) && $last['msg']==='RETRY') {
        $http = 502;
    } elseif (isset($last['msg']) && $last['msg']==='NO_DATA') {
        $http = 504;
    } elseif (isset($last['http_code'])) {
        $http = (int)$last['http_code'];
    }
    $last['_cache'] = 'MISS';
    json_out($last, $http);
}
json_out(array('error'=>'비정상 응답','http_code'=>500,'_cache'=>'MISS'),500);
